🏀 NBA Prediction System - Command Reference

================================================================================
CORE COMMANDS
================================================================================

# Complete analysis (Moneyline + Totals + Confluence) - RECOMMENDED
python predictions.py
python predictions.py --tomorrow
python predictions.py --date 2026-01-25

# Individual prediction types
python predictions.py --moneyline       # Moneyline only
python predictions.py --totals          # Totals only

# Options
python predictions.py --show-all        # Show all games
python predictions.py --min-edge 0.08   # Conservative (8% min edge)
python predictions.py --max-bets 3      # Limit to 3 bets
python predictions.py --bankroll 5000   # Custom bankroll
python predictions.py --typical-total 225.0  # Custom O/U line

================================================================================
TRAINING COMMANDS
================================================================================

# Train moneyline model
python main.py --days-back 30

# Train totals model
python train_totals_model.py

# Full pipeline (fetch data + train + predict)
python main.py

# Advanced training
python main.py --tune                  # Hyperparameter tuning
python main.py --backtest              # Run backtesting
python main.py --no-update             # Skip data download
python main.py --days-back 60          # More training data

================================================================================
PERFORMANCE TRACKING
================================================================================

# Update results
python update_results.py --days 1      # Update yesterday
python update_results.py --days 7      # Weekly review
python update_results.py --export      # Export to CSV
python update_results.py --no-report   # Update without report

================================================================================
